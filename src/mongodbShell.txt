
db.createCollection("books", {
  validator: {
    $jsonSchema: {
      bsonType: "object",
      required: ["title"],
      properties: {
        title: {
          bsonType: "string",
          minLength: 1
        }
      }
    }
  }
})

db.authors.insertOne({
  name: "George Orwell",
  country: "United Kingdom"
})

db.createCollection("logs", {
  capped: true,
  size: 1048576
})

db.books.createIndex({
  title: 1
})

db.books.insertOne({
  title: "Future",
  author: "Aldous Huxley",
  year: 2021,
  genres: ["Science Fiction", "Dystopian"]
})

db.books.insertMany([
  {
    title: "Brave New World",
    author: "Aldous Huxley",
    year: 1932,
    genres: ["Science Fiction", "Dystopian"]
  },
  {
    title: "The Hobbit",
    author: "J.R.R. Tolkien",
    year: 1937,
    genres: ["Fantasy", "Adventure"]
  },
  {
    title: "The Road",
    author: "Cormac McCarthy",
    year: 2006,
    genres: ["Dystopian", "Drama"]
  },
  {
    title: "Dune",
    author: "Frank Herbert",
    year: 1965,
    genres: ["Science Fiction", "Adventure"]
  }
])

db.logs.insertOne({
  action: "Book inserted",
  bookTitle: "Future",
  timestamp: new Date()
})

db.books.updateOne(
  { title: "Future" },
  { $set: { year: 2022 } }
)

db.books.find({
  title: "Brave New World"
})

db.books.find({
  year: {
    $gte: 1990,
    $lte: 2010
  }
})

db.books.find({
  genres: "Science Fiction"
})

db.books.find()
  .sort({ year: -1 })
  .skip(2)
  .limit(3)

db.books.find({
  year: {
    $type: "int"
  }
})

db.books.find({
  genres: {
    $nin: ["Horror", "Science Fiction"]
  }
})

db.books.deleteMany({
  year: {
    $lt: 2000
  }
})

db.books.aggregate([
  {
    $match: {
      year: {
        $gt: 2000
      }
    }
  },
  {
    $sort: {
      year: -1
    }
  }
])

db.books.aggregate([
  {
    $match: {
      year: {
        $gt: 2000
      }
    }
  },
  {
    $project: {
      _id: 0,
      title: 1,
      author: 1,
      year: 1
    }
  }
])

db.books.aggregate([
  {
    $unwind: "$genres"
  }
])

db.books.aggregate([
  {
    $lookup: {
      from: "logs",
      localField: "title",
      foreignField: "bookTitle",
      as: "logs"
    }
  }
])
